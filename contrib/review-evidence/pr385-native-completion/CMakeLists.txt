cmake_minimum_required(VERSION 3.22)
project(RemoteWorkCompletion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
set(BTX_SOURCE_ROOT "" CACHE PATH "Exact385 source")
set(CRYPTO_ARCHIVE "" CACHE FILEPATH "Identified native crypto archive")
set(JSON_ARCHIVE "" CACHE FILEPATH "Identified native JSON archive")
include_directories(common/src ${BTX_SOURCE_ROOT}/src ${BTX_SOURCE_ROOT}/src/univalue/include ${BTX_SOURCE_ROOT}/src/libbitcoinpqc/include)
file(GLOB CORE common/src/modelnet/security_v2/*.cpp)
list(FILTER CORE EXCLUDE REGEX "/(main|service|work)\\.cpp$")
add_library(core OBJECT ${CORE})
target_link_libraries(core PRIVATE SQLite::SQLite3 Threads::Threads)
target_compile_options(core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(harness OBJECT security_tests.cpp)
target_link_libraries(harness PRIVATE SQLite::SQLite3 Threads::Threads)
foreach(variant baseline candidate)
 add_executable(${variant} ${variant}/service.cpp ${variant}/work.cpp $<TARGET_OBJECTS:core> $<TARGET_OBJECTS:harness>)
 target_include_directories(${variant} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/${variant}/include)
 target_link_libraries(${variant} PRIVATE ${CRYPTO_ARCHIVE} ${JSON_ARCHIVE} SQLite::SQLite3 Threads::Threads)
 target_link_options(${variant} PRIVATE -Wl,-z,stack-size=8388608 "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${variant}.map")
endforeach()
add_executable(security-probe ${BTX_SOURCE_ROOT}/test/util/model_security/sandbox_probe.c)
target_link_options(security-probe PRIVATE -static)
