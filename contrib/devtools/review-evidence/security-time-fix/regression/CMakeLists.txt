cmake_minimum_required(VERSION 3.22)
project(SecurityTimeFixPortable LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This profile requires native Linux and the real static worker")
endif()
set(BTX_SOURCE_ROOT "" CACHE PATH "Read-only exact fd3a905f checkout/archive")
if(NOT BTX_SOURCE_ROOT)
  message(FATAL_ERROR "Set BTX_SOURCE_ROOT to the validated exact source tree")
endif()
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
set(BTX_SECURITY_OVERLAY "${CMAKE_CURRENT_SOURCE_DIR}/product" CACHE PATH "Assembled timing product" FORCE)
set(BTX_INTEGRATED_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/no-daemon" CACHE PATH "No daemon targets" FORCE)
# Rebuild only the official standalone dependency targets; no binary archives ship.
add_subdirectory("${BTX_SOURCE_ROOT}/test/util/model_security" upstream EXCLUDE_FROM_ALL)
set_target_properties(security-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include_directories(test-source/common/src ${BTX_SOURCE_ROOT}/src ${BTX_SOURCE_ROOT}/src/univalue/include ${BTX_SOURCE_ROOT}/src/libbitcoinpqc/include)
file(GLOB COMMON test-source/common/src/modelnet/security_v2/*.cpp)
list(FILTER COMMON EXCLUDE REGEX "/(main|evaluation|trust|service)\\.cpp$")
add_library(security_common OBJECT ${COMMON})
add_library(harness OBJECT harness/security_tests.cpp)
foreach(variant baseline candidate)
  add_library(core-${variant} OBJECT test-source/${variant}/trust.cpp test-source/${variant}/evaluation.cpp)
endforeach()
foreach(target security_common harness core-baseline core-candidate)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(${target} PRIVATE SQLite::SQLite3 Threads::Threads)
endforeach()
foreach(variant baseline candidate combined)
  if(variant STREQUAL "combined")
    set(CORE candidate)
  else()
    set(CORE ${variant})
  endif()
  add_executable(time-${variant} test-source/${variant}/service.cpp $<TARGET_OBJECTS:security_common> $<TARGET_OBJECTS:harness> $<TARGET_OBJECTS:core-${CORE}>)
  add_dependencies(time-${variant} security-probe)
  target_compile_options(time-${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_options(time-${variant} PRIVATE -Wl,-z,stack-size=8388608 "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/time-${variant}.map")
  target_link_libraries(time-${variant} PRIVATE btx_security_crypto btx_security_json SQLite::SQLite3 Threads::Threads)
endforeach()
