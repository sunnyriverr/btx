cmake_minimum_required(VERSION 3.22)
project(SecurityTimeFix LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
set(BTX_DEPENDENCY_SOURCE <RECORDED_DEPENDENCY_SOURCE> CACHE PATH "Read-only dependency source matching fd3a905f")
include_directories(test-source/common/src ${BTX_DEPENDENCY_SOURCE}/src ${BTX_DEPENDENCY_SOURCE}/src/univalue/include ${BTX_DEPENDENCY_SOURCE}/src/libbitcoinpqc/include)
file(GLOB COMMON test-source/common/src/modelnet/security_v2/*.cpp)
list(FILTER COMMON EXCLUDE REGEX "/(main|evaluation|trust|service)\\.cpp$")
add_library(security_common OBJECT ${COMMON})
add_library(harness OBJECT harness/security_tests.cpp)
foreach(variant baseline candidate)
 add_library(core-${variant} OBJECT test-source/${variant}/trust.cpp test-source/${variant}/evaluation.cpp)
endforeach()
foreach(target security_common harness core-baseline core-candidate)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_libraries(${target} PRIVATE SQLite::SQLite3 Threads::Threads)
endforeach()
foreach(variant baseline candidate combined)
 if(variant STREQUAL "combined")
  set(CORE candidate)
 else()
  set(CORE ${variant})
 endif()
 add_executable(time-${variant} test-source/${variant}/service.cpp $<TARGET_OBJECTS:security_common> $<TARGET_OBJECTS:harness> $<TARGET_OBJECTS:core-${CORE}>)
 target_compile_options(time-${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 target_link_options(time-${variant} PRIVATE -Wl,-z,stack-size=8388608 "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/time-${variant}.map")
 target_link_libraries(time-${variant} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/dependencies/crypto.a ${CMAKE_CURRENT_SOURCE_DIR}/dependencies/json.a SQLite::SQLite3 Threads::Threads)
endforeach()
add_executable(security-probe product/test/util/model_security/sandbox_probe.c)
target_link_options(security-probe PRIVATE -static)
