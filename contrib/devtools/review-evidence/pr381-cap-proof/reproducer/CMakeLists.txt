cmake_minimum_required(VERSION 3.22)
project(CapProof LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
set(BTX_SOURCE_ROOT "" CACHE PATH "Exact381 source")
set(CRYPTO_ARCHIVE "" CACHE FILEPATH "Identified native crypto archive")
set(JSON_ARCHIVE "" CACHE FILEPATH "Identified native JSON archive")
include_directories(src ${BTX_SOURCE_ROOT}/src ${BTX_SOURCE_ROOT}/src/univalue/include ${BTX_SOURCE_ROOT}/src/libbitcoinpqc/include)
file(GLOB CORE src/modelnet/security_v2/*.cpp)
list(FILTER CORE EXCLUDE REGEX "/(main|service)\\.cpp$")
add_library(core OBJECT ${CORE})
target_link_libraries(core PRIVATE SQLite::SQLite3 Threads::Threads)
target_compile_options(core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(harness OBJECT test/util/model_security/security_tests.cpp)
target_link_libraries(harness PRIVATE SQLite::SQLite3 Threads::Threads)
foreach(cap 2 99)
 if(cap EQUAL 2)
  set(SERVICE src/modelnet/security_v2/service.cpp)
 else()
  set(SERVICE cap99-service.cpp)
 endif()
 add_executable(cap${cap} ${SERVICE} $<TARGET_OBJECTS:core> $<TARGET_OBJECTS:harness>)
 target_link_libraries(cap${cap} PRIVATE ${CRYPTO_ARCHIVE} ${JSON_ARCHIVE} SQLite::SQLite3 Threads::Threads)
 target_link_options(cap${cap} PRIVATE -Wl,-z,stack-size=8388608 "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/cap${cap}.map")
endforeach()
add_executable(security-probe ${BTX_SOURCE_ROOT}/test/util/model_security/sandbox_probe.c)
target_link_options(security-probe PRIVATE -static)
